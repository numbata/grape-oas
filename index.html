catalog_api.rbRUBY
require "grape_oas"

class CatalogAPI < Grape::API
  format :json

  desc "Find a book"
  params do
    requires :id, type: Integer, desc: "Book ID"
  end
  get "books/:id" do
    { id: params[:id], title: "The Ruby Programming Language" }
  end

  add_oas_documentation(
    info: { title: "Catalog API", version: "1.0.0" },
  )
end

FROM ROUTES TO SCHEMAS

Already in your API.
Now in your docs.

grape-oas reads the metadata you declare in Grape and builds a shared API model. Separate exporters turn that model into the OpenAPI format you need.
